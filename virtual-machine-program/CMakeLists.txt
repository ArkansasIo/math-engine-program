cmake_minimum_required(VERSION 3.25)
project(VirtualMachine1024 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(vm1024 src/cpu1024/CPU1024.cpp src/gpu1024/GPU1024.cpp src/memory/Memory.cpp src/machine/Decoder.cpp src/quantum_cpu/QubitRegister.cpp src/quantum_cpu/QuantumCircuit.cpp src/quantum_cpu/Gates.cpp src/quantum_cpu/Measurement.cpp src/math_bridge/MathBridge.cpp)
target_include_directories(vm1024 PUBLIC include)
if(MSVC)
 target_compile_options(vm1024 PRIVATE /W4 /permissive-)
else()
 target_compile_options(vm1024 PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(vm1024-cli src/main.cpp)
target_link_libraries(vm1024-cli PRIVATE vm1024)
if(WIN32)
 add_executable(vm1024-dashboard WIN32 src/ui/main_win32.cpp src/ui/Virtual1024App.cpp)
 target_include_directories(vm1024-dashboard PRIVATE include)
 target_link_libraries(vm1024-dashboard PRIVATE vm1024 user32 gdi32)
endif()
add_executable(vm1024-smoke tests/test_smoke.cpp)
target_link_libraries(vm1024-smoke PRIVATE vm1024)
add_executable(vm1024-decoder-test tests/test_decoder.cpp)
target_link_libraries(vm1024-decoder-test PRIVATE vm1024)
add_executable(vm1024-quantum-test tests/test_quantum.cpp)
target_link_libraries(vm1024-quantum-test PRIVATE vm1024)
enable_testing()
add_test(NAME vm1024_smoke COMMAND vm1024-smoke)
add_test(NAME vm1024_decoder COMMAND vm1024-decoder-test)
add_test(NAME vm1024_quantum COMMAND vm1024-quantum-test)
